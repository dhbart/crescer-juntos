package dhbart.crescerjuntos.domain.model;

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Objects;

public class Familia {
    
    private final Long id;
    private String nome;
    private boolean ativa;
    private final LocalDateTime criadoEm;
    private LocalDateTime atualizadoEm;
    private final List<Crianca> criancas;
    
    public Familia(String nome) {
        this.nome = Objects.requireNonNull(nome, "Nome da família é obrigatório");
        if (nome.isBlank()) {
            throw new IllegalArgumentException("Nome da família não pode ser vazio");
        }
        this.ativa = true;
        this.criadoEm = LocalDateTime.now();
        this.criancas = new ArrayList<>();
    }
    
    public Familia(Long id, String nome, boolean ativa, LocalDateTime criadoEm, LocalDateTime atualizadoEm) {
        this.id = id;
        this.nome = nome;
        this.ativa = ativa;
        this.criadoEm = criadoEm;
        this.atualizadoEm = atualizadoEm;
        this.criancas = new ArrayList<>();
    }
    
    public void adicionarCrianca(Crianca crianca) {
        Objects.requireNonNull(crianca, "Criança não pode ser nula");
        if (criancas.contains(crianca)) {
            throw new IllegalArgumentException("Criança já pertence a esta família");
        }
        this.criancas.add(crianca);
    }
    
    public void removerCrianca(Crianca crianca) {
        if (!criancas.remove(crianca)) {
            throw new IllegalArgumentException("Criança não encontrada nesta família");
        }
    }
    
    public void desativar() {
        this.ativa = false;
        this.atualizadoEm = LocalDateTime.now();
    }
    
    public void ativar() {
        this.ativa = true;
        this.atualizadoEm = LocalDateTime.now();
    }
    
    public void atualizarNome(String novoNome) {
        if (novoNome == null || novoNome.isBlank()) {
            throw new IllegalArgumentException("Nome da família não pode ser vazio");
        }
        this.nome = novoNome;
        this.atualizadoEm = LocalDateTime.now();
    }
    
    public boolean isAtiva() {
        return ativa;
    }
    
    // Getters
    public Long getId() { return id; }
    public String getNome() { return nome; }
    public LocalDateTime getCriadoEm() { return criadoEm; }
    public LocalDateTime getAtualizadoEm() { return atualizadoEm; }
    public List<Crianca> getCriancas() { return Collections.unmodifiableList(criancas); }
    
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Familia familia = (Familia) o;
        return id != null && id.equals(familia.id);
    }
    
    @Override
    public int hashCode() {
        return getClass().hashCode();
    }
}

package dhbart.crescerjuntos.domain.model;

import java.time.DayOfWeek;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Objects;

public class Tarefa {
    
    private final Long id;
    private String titulo;
    private String descricao;
    private Frequencia frequencia;
    private List<DayOfWeek> diasDaSemana;
    private Integer diaDoMes;
    private LocalDate dataInicio;
    private LocalDate dataFim;
    private LocalTime horarioPreferido;
    private int pontos;
    private boolean ativa;
    private Familia familia;
    private final LocalDateTime criadoEm;
    private LocalDateTime atualizadoEm;
    
    public Tarefa(String titulo, Frequencia frequencia, Familia familia, int pontos) {
        this.titulo = Objects.requireNonNull(titulo, "Título é obrigatório");
        if (titulo.isBlank()) {
            throw new IllegalArgumentException("Título não pode ser vazio");
        }
        this.frequencia = Objects.requireNonNull(frequencia, "Frequência é obrigatória");
        this.familia = Objects.requireNonNull(familia, "Família é obrigatória);
        this.pontos = validarPontos(pontos);
        this.diasDaSemana = new ArrayList<>();
        this.ativa = true;
        this.criadoEm = LocalDateTime.now();
    }
    
    private int validarPontos(int pontos) {
        if (pontos < 0) {
            throw new IllegalArgumentException("Pontos não podem ser negativos");
        }
        if (pontos > 100) {
            throw new IllegalArgumentException("Pontos não podem exceder 100 por tarefa");
        }
        return pontos;
    }
    
    public void configurarDiasDaSemana(List<DayOfWeek> dias) {
        if (this.frequencia != Frequencia.SEMANAL) {
            throw new IllegalStateException("Dias da semana só podem ser configurados para frequência SEMANAL");
        }
        if (dias == null || dias.isEmpty()) {
            throw new IllegalArgumentException("Deve selecionar pelo menos um dia da semana");
        }
        this.diasDaSemana = new ArrayList<>(dias);
        this.atualizadoEm = LocalDateTime.now();
    }
    
    public void configurarDiaDoMes(int dia) {
        if (this.frequencia != Frequencia.MENSAL) {
            throw new IllegalStateException("Dia do mês só pode ser configurado para frequência MENSAL");
        }
        if (dia < 1 || dia > 31) {
            throw new IllegalArgumentException("Dia do mês deve ser entre 1 e 31");
        }
        this.diaDoMes = dia;
        this.atualizadoEm = LocalDateTime.now();
    }
    
    public void configurarPeriodoPersonalizado(LocalDate inicio, LocalDate fim) {
        if (this.frequencia != Frequencia.PERIODO_PERSONALIZADO) {
            throw new IllegalStateException("Período só pode ser configurado para frequência PERIODO_PERSONALIZADO");
        }
        if (inicio == null || fim == null) {
            throw new IllegalArgumentException("Data de início e fim são obrigatórias");
        }
        if (fim.isBefore(inicio)) {
            throw new IllegalArgumentException("Data de fim não pode ser anterior à data de início");
        }
        this.dataInicio = inicio;
        this.dataFim = fim;
        this.atualizadoEm = LocalDateTime.now();
    }
    
    public booleanEstaAtivaNaData(LocalDate data) {
        if (!ativa) return false;
        if (data == null) return false;
        
        // Verificar período personalizado
        if (frequencia == Frequencia.PERIODO_PERSONALIZADO) {
            if (dataInicio == null || dataFim == null) return false;
            return !data.isBefore(dataInicio) && !data.isAfter(dataFim);
        }
        
        // Verificar data início/fim geral
        if (dataInicio != null && data.isBefore(dataInicio)) return false;
        if (dataFim != null && data.isAfter(dataFim)) return false;
        
        return switch (frequencia) {
            case DIARIA -> true;
            case SEMANAL -> diasDaSemana.contains(data.getDayOfWeek());
            case MENSAL -> diaDoMes != null && data.getDayOfMonth() == diaDoMes;
            default -> false;
        };
    }
    
    public void desativar() {
        this.ativa = false;
        this.atualizadoEm = LocalDateTime.now();
    }
    
    public void ativar() {
        this.ativa = true;
        this.atualizadoEm = LocalDateTime.now();
    }
    
    public void atualizarPontos(int novosPontos) {
        this.pontos = validarPontos(novosPontos);
        this.atualizadoEm = LocalDateTime.now();
    }
    
    // Getters
    public Long getId() { return id; }
    public String getTitulo() { return titulo; }
    public String getDescricao() { return descricao; }
    public Frequencia getFrequencia() { return frequencia; }
    public List<DayOfWeek> getDiasDaSemana() { return Collections.unmodifiableList(diasDaSemana); }
    public Integer getDiaDoMes() { return diaDoMes; }
    public LocalDate getDataInicio() { return dataInicio; }
    public LocalDate getDataFim() { return dataFim; }
    public LocalTime getHorarioPreferido() { return horarioPreferido; }
    public int getPontos() { return pontos; }
    public boolean isAtiva() { return ativa; }
    public Familia getFamilia() { return familia; }
    public LocalDateTime getCriadoEm() { return criadoEm; }
    public LocalDateTime getAtualizadoEm() { return atualizadoEm; }
    
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Tarefa tarefa = (Tarefa) o;
        return id != null && id.equals(tarefa.id);
    }
    
    @Override
    public int hashCode() {
        return getClass().hashCode();
    }
}

package dhbart.crescerjuntos.domain.model;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Objects;

public class ExecucaoTarefa {
    
    private static final int PONTOS_MAXIMOS = 100;
    
    private final Long id;
    private Tarefa tarefa;
    private Crianca crianca;
    private LocalDate dataExecucao;
    private StatusExecucao status;
    private int pontosAplicados;
    private String observacao;
    private LocalDateTime registradoEm;
    private LocalDateTime avaliadoEm;
    private final LocalDateTime criadoEm;
    
    public ExecucaoTarefa(Tarefa tarefa, Crianca crianca, LocalDate dataExecucao) {
        this.tarefa = Objects.requireNonNull(tarefa, "Tarefa é obrigatória");
        this.crianca = Objects.requireNonNull(crianca, "Criança é obrigatória");
        this.dataExecucao = Objects.requireNonNull(dataExecucao, "Data de execução é obrigatória");
        this.status = StatusExecucao.PENDENTE;
        this.pontosAplicados = 0;
        this.criadoEm = LocalDateTime.now();
    }
    
    public void registrar(String observacao) {
        if (this.status != StatusExecucao.PENDENTE) {
            throw new IllegalStateException("Só é possível registrar execuções pendentes");
        }
        this.status = StatusExecucao.REGISTRADA;
        this.observacao = observacao;
        this.registradoEm = LocalDateTime.now();
    }
    
    public void aprovar(int pontos) {
        if (this.status != StatusExecucao.REGISTRADA) {
            throw new IllegalStateException("Só é possível aprovar execuções registradas");
        }
        if (pontos < 0 || pontos > PONTOS_MAXIMOS) {
            throw new IllegalArgumentException("Pontos devem estar entre 0 e " + PONTOS_MAXIMOS);
        }
        this.status = StatusExecucao.APROVADA;
        this.pontosAplicados = pontos;
        this.avaliadoEm = LocalDateTime.now();
    }
    
    public void rejeitar(String motivo) {
        if (this.status != StatusExecucao.REGISTRADA) {
            throw new IllegalStateException("Só é possível rejeitar execuções registradas");
        }
        this.status = StatusExecucao.REJEITADA;
        this.observacao = motivo;
        this.avaliadoEm = LocalDateTime.now();
    }
    
    public void expirar() {
        if (this.status != StatusExecucao.PENDENTE) {
            throw new IllegalStateException("Só é possível expirar execuções pendentes");
        }
        this.status = StatusExecucao.EXPIRADA;
    }
    
    public boolean isPendente() {
        return this.status == StatusExecucao.PENDENTE;
    }
    
    public boolean isAprovada() {
        return this.status == StatusExecucao.APROVADA;
    }
    
    // Getters
    public Long getId() { return id; }
    public Tarefa getTarefa() { return tarefa; }
    public Crianca getCrianca() { return crianca; }
    public LocalDate getDataExecucao() { return dataExecucao; }
    public StatusExecucao getStatus() { return status; }
    public int getPontosAplicados() { return pontosAplicados; }
    public String getObservacao() { return observacao; }
    public LocalDateTime getRegistradoEm() { return registradoEm; }
    public LocalDateTime getAvaliadoEm() { return avaliadoEm; }
    public LocalDateTime getCriadoEm() { return criadoEm; }
    
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        ExecucaoTarefa that = (ExecucaoTarefa) o;
        return id != null && id.equals(that.id);
    }
    
    @Override
    public int hashCode() {
        return getClass().hashCode();
    }
}

package dhbart.crescerjuntos.infrastructure.persistence.entity;

import jakarta.persistence.*;
import java.time.LocalDateTime;

@Entity
@Table(name = "familias")
public class FamiliaEntity {

    @GeneratedValue(strategy = GenerationType.IDENTITY)
